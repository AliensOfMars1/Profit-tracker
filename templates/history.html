@app.route('/delete-calculation/', methods=['POST']) @login_required def delete_calculation(id): try: calc = Calculation.query.get_or_404(id) calc_date = calc.created_at db.session.delete(calc) db.session.commit() flash(f'Calculation from {calc_date.strftime("%B %d, %Y")} deleted successfully!', 'success') except Exception as e: flash(f'Error deleting calculation: {str(e)}', 'error') return redirect(url_for('history'))
